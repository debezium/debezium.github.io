{%- comment -%} Version entries generated for the navigation drop-downs. {% include dbz/nav-versions.html kind="documentation" style="desktop" %} kind - "documentation" links into the reference docs, "releases" into the release pages style - "desktop" for the drop-down panel, "drawer" for the mobile menu Newest series first. The first stable series encountered is the current one and is labelled accordingly; anything not stable is a development series. The documentation list starts with the nightly docs, which are built from the main branch and published under /documentation/reference/nightly/. Nightly has no release page, so the releases list does not include it. {%- endcomment -%} {%- if include.kind == "documentation" -%} {% include dbz/nav-version-link.html version="nightly" href="/documentation/reference/nightly/" label="nightly" stable=false style=include.style %} {%- endif -%} {%- assign versionOrder = site.data.versions.versions | reverse -%} {%- assign seenStable = false -%} {%- for version in versionOrder -%} {%- assign formatedVersion = version | replace: ".", "" -%} {%- assign releaseDetails = site.data.releases[formatedVersion] -%} {%- if releaseDetails.series.displayed -%} {%- assign sortedReleaseDetails = releaseDetails | sort -%} {%- assign lastIndex = sortedReleaseDetails.size | minus: 2 -%} {%- assign formattedLastVersion = sortedReleaseDetails[lastIndex][0] -%} {%- assign latestRelease = releaseDetails[formattedLastVersion] -%} {%- if latestRelease.stable -%} {%- assign isStable = true -%} {%- if seenStable -%} {%- assign labelText = "stable" -%} {%- else -%} {%- assign labelText = "latest stable" -%} {%- assign seenStable = true -%} {%- endif -%} {%- else -%} {%- assign isStable = false -%} {%- assign labelText = "development" -%} {%- endif -%} {%- if include.kind == "releases" -%} {%- assign href = version | prepend: "/releases/" | append: "/" -%} {%- else -%} {%- assign href = version | prepend: "/documentation/reference/" | append: "/" -%} {%- endif -%} {% include dbz/nav-version-link.html version=version href=href label=labelText stable=isStable style=include.style %} {%- endif -%} {%- endfor -%}
